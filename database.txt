DROP TABLE IF EXISTS PRODUCT;

DROP TABLE IF EXISTS SELLER;
DROP TABLE IF EXISTS BUYER;
DROP TABLE IF EXISTS PERSON;

-- =============================================================================
-- DDL: TABLE CREATION
-- =============================================================================

-- 1. PERSON (Superclass)
CREATE TABLE PERSON (
    PersonID INT PRIMARY KEY,
    Email VARCHAR(255) UNIQUE NOT NULL,
    PasswordHash VARCHAR(255) NOT NULL,
    FirstName VARCHAR(100) NOT NULL,
    LastName VARCHAR(100) NOT NULL,
    AddressLine1 VARCHAR(255),
    City VARCHAR(100),
    State VARCHAR(50),
    PostalCode VARCHAR(20),
    CreatedDate TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. BUYER (Subclass of Person - Disjoint EER)
CREATE TABLE BUYER (
    BuyerID INT PRIMARY KEY,
    MembershipStatus VARCHAR(50),
    CONSTRAINT fk_buyer_person FOREIGN KEY (BuyerID) REFERENCES PERSON(PersonID) ON DELETE CASCADE
);

-- 3. SELLER (Subclass of Person - Disjoint EER)
CREATE TABLE SELLER (
    SellerID INT PRIMARY KEY,
    StoreName VARCHAR(100) NOT NULL,
    Rating DECIMAL(3,2),
    CONSTRAINT fk_seller_person FOREIGN KEY (SellerID) REFERENCES PERSON(PersonID) ON DELETE CASCADE
);

-- 4. SUPPORTAGENT (Subclass of Person - Disjoint EER)
CREATE TABLE SUPPORTAGENT (
    AgentID INT PRIMARY KEY,
    HireDate DATE,
    Role VARCHAR(100),
    CONSTRAINT fk_supportagent_person FOREIGN KEY (AgentID) REFERENCES PERSON(PersonID) ON DELETE CASCADE
);

-- 5. CATEGORY
CREATE TABLE CATEGORY (
    CategoryID INT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL
);

-- 6. PRODUCT
CREATE TABLE PRODUCT (
    ProductID INT PRIMARY KEY,
    SellerID INT NOT NULL,
    CategoryID INT NOT NULL,
    Name VARCHAR(255) NOT NULL,
    Description TEXT,
    BasePrice DECIMAL(10,2) NOT NULL,
    CONSTRAINT fk_product_seller FOREIGN KEY (SellerID) REFERENCES SELLER(SellerID),
    CONSTRAINT fk_product_category FOREIGN KEY (CategoryID) REFERENCES CATEGORY(CategoryID)
);

-- 7. COURIER
CREATE TABLE COURIER (
    CourierID INT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    ContactPhone VARCHAR(50)
);

-- 8. SHIPMENT
CREATE TABLE SHIPMENT (
    ShipmentID INT PRIMARY KEY,
    CourierID INT NOT NULL,
    TrackingNumber VARCHAR(100) UNIQUE NOT NULL,
    ShippedDate TIMESTAMP,
    ExpectedDeliveryDate TIMESTAMP,
    CONSTRAINT fk_shipment_courier FOREIGN KEY (CourierID) REFERENCES COURIER(CourierID)
);

-- 9. ORDER_TABLE (Named ORDER_TABLE to avoid reserved keyword conflicts)
CREATE TABLE ORDER_TABLE (
    OrderID INT PRIMARY KEY,
    BuyerID INT NOT NULL,
    ShipmentID INT,
    OrderDate TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    DeliveryStatus VARCHAR(50) NOT NULL,
    TotalAmount DECIMAL(10,2) NOT NULL,
    CONSTRAINT fk_order_buyer FOREIGN KEY (BuyerID) REFERENCES BUYER(BuyerID),
    CONSTRAINT fk_order_shipment FOREIGN KEY (ShipmentID) REFERENCES SHIPMENT(ShipmentID)
);

-- 10. ORDERITEM
CREATE TABLE ORDERITEM (
    OrderItemID INT,
    OrderID INT,
    ProductID INT NOT NULL,
    Quantity INT NOT NULL,
    PriceAtPurchase DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (OrderItemID, OrderID),
    CONSTRAINT fk_orderitem_order FOREIGN KEY (OrderID) REFERENCES ORDER_TABLE(OrderID) ON DELETE CASCADE,
    CONSTRAINT fk_orderitem_product FOREIGN KEY (ProductID) REFERENCES PRODUCT(ProductID)
);

-- 11. TRACKINGEVENT
CREATE TABLE TRACKINGEVENT (
    TrackingEventID INT PRIMARY KEY,
    ShipmentID INT NOT NULL,
    Timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    Location VARCHAR(255),
    StatusDescription VARCHAR(255),
    CONSTRAINT fk_tracking_shipment FOREIGN KEY (ShipmentID) REFERENCES SHIPMENT(ShipmentID) ON DELETE CASCADE
);

-- 12. PAYMENT
CREATE TABLE PAYMENT (
    PaymentID INT PRIMARY KEY,
    OrderID INT NOT NULL,
    PaymentDate TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    Amount DECIMAL(10,2) NOT NULL,
    PaymentMethod VARCHAR(50) NOT NULL,
    CONSTRAINT fk_payment_order FOREIGN KEY (OrderID) REFERENCES ORDER_TABLE(OrderID) ON DELETE CASCADE
);

-- 13. REVIEW
CREATE TABLE REVIEW (
    ReviewID INT PRIMARY KEY,
    OrderID INT NOT NULL,
    Rating INT CHECK (Rating BETWEEN 1 AND 5),
    Comment TEXT,
    ReviewDate TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_review_order FOREIGN KEY (OrderID) REFERENCES ORDER_TABLE(OrderID) ON DELETE CASCADE
);

-- 14. RETURNREQUEST
CREATE TABLE RETURNREQUEST (
    ReturnID INT PRIMARY KEY,
    OrderID INT NOT NULL,
    Reason TEXT NOT NULL,
    Status VARCHAR(50) NOT NULL,
    RequestedDate TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_return_order FOREIGN KEY (OrderID) REFERENCES ORDER_TABLE(OrderID) ON DELETE CASCADE
);
